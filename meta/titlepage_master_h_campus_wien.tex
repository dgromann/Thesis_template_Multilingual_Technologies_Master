% -- Commands for the title page --------------

% -- Title of the work
\newcommand{\TitleValue}{Title}
\newcommand{\SubtitleValue}{Subtitle}

% -- Name
\newcommand{\NameValue}{\textgreater degree(s) first name family name degree(s)\textless \par }


% -- Desired degree
\newcommand{\DegreeValue}{Master of Science (MSc \hspace{0.1cm}(MSC)}

% -- Year
\newcommand{\YearValue}{\textgreater Year\textless }

% -- Degree programme name
\newcommand{\ProgrammeNameValue}{Multilingual Technologies}

% -- Supervisors
\newcommand{\SupervisorValue}{\textgreater degree first name family name \textless}
\newcommand{\CoSupervisorValue}{}

% -- Commands to update the values
\newcommand{\Title}[1]{\renewcommand{\TitleValue}{#1}}
\newcommand{\Subtitle}[1]{\renewcommand{\SubtitleValue}{#1}}
\newcommand{\Name}[1]{\renewcommand{\NameValue}{#1}}
\newcommand{\Degree}[1]{\renewcommand{\DegreeValue}{#1} }

\newcommand{\Supervisor}[1]{\renewcommand{\SupervisorValue}{#1}}
\newcommand{\CoSupervisor}[1]{\renewcommand{\CoSupervisorValue}{#1}}
\newcommand{\CoSupValue}{1}
